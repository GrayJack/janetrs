28/// The possible errors for the [`JanetClient`].
29#[derive(Debug, Clone, Copy, PartialEq, PartialOrd, Eq, Ord, Hash)]
30#[non_exhaustive]
31pub enum Error {
32    /// May happen when trying to initialize two or more [`JanetClient`].
33    AlreadyInit,
34    /// May happen when trying to run code that does not compile.
35    CompileError,
36    /// May happen when trying to run a Janet code without a environment table.
37    EnvNotInit,
38    /// May happen when trying to run code that failed to be parsed.
39    ParseError,
40    /// May happen when the VM errors while running code.
41    RunError,
42}
43
44impl Display for Error {
45    #[inline]
46    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
47        match self {
48            Self::AlreadyInit => f.pad("Janet client already initialized"),
49            Self::CompileError => f.pad("Failed to compile code"),
50            Self::EnvNotInit => f.pad("The environment table was not initialized"),
51            Self::ParseError => f.pad("Failed to parse code"),
52            Self::RunError => f.pad("Runtime VM error"),
53        }
54    }
55}

61/// Janet client that initialize the Janet runtime.
62///
63/// If in a `no_std` environment you can only initialize the runtime through the safe
64/// interface only once, since the static atomic global cannot be thread local in a
65/// `no_std` environment, if you're on a multi-thread + `no_std` environment refer to use
66/// [`init_unchecked`].
67///
68/// [`init_unchecked`]: ./struct.JanetClient.html#method.init_unchecked.html
69#[derive(Debug)]
70pub struct JanetClient {
71    env_table: Option<JanetEnvironment>,
72}
73
74impl JanetClient {
75    /// Initialize Janet global state.
76    ///
77    /// This must be called only once per thread if using Janet in a multithreaded
78    /// environment, as all Janet global state is thread local by default.
79    ///
80    /// If tried to initialize the client more than once it returns a `Err` variant.
81    #[inline]
82    pub fn init() -> Result<Self, Error> {
83        #[cfg(feature = "std")]
84        let init_state = INIT.with(|i| i.swap(true, Ordering::SeqCst));
85        #[cfg(not(feature = "std"))]
86        let init_state = INIT.swap(true, Ordering::SeqCst);
87
88        if init_state {
89            return Err(Error::AlreadyInit);
90        }
91
92        // SAFETY: We use a static AtomicBool to make sure that it is started only once (per
93        // thread if "std" feature activated)
94        unsafe { evil_janet::janet_init() };
95        Ok(Self { env_table: None })
96    }
97
98    /// Initialize Janet global state without checking.
99    ///
100    /// This must be called only once per thread if using Janet in a multithreaded
101    /// environment, as all Janet global state is thread local by default.
102    ///
103    /// # Safety
104    /// If initialized more than once per thread, and more than one drop, you can have a
105    /// double free, if one drop and another continue to execute, it will crash with
106    /// segmentation fault.
107    #[inline]
108    #[must_use = "function is a constructor associated function"]
109    pub unsafe fn init_unchecked() -> Self {
110        evil_janet::janet_init();
111        Self { env_table: None }
112    }

114    /// Initialize Janet global state and load the default environment of Janet.
115    ///
116    /// This must be called only once per thread if using Janet in a multithreaded
117    /// environment, as all Janet global state is thread local by default.
118    ///
119    /// If tried to initialize the client more than once it returns a `Err` variant.
120    ///
121    /// The default environment of Janet contains all the Janet C code as well as the
122    /// code in [`boot.janet`](https://github.com/janet-lang/janet/blob/master/src/boot/boot.janet).
123    #[inline]
124    pub fn init_with_default_env() -> Result<Self, Error> {
125        let mut client = Self::init()?;
126        client.env_table = Some(JanetEnvironment::default());
127        Ok(client)

174    /// Add a Janet immutable variable to the client environment if it has one, otherwise
175    /// creates it with the default one.
176    ///
177    /// # Examples
178    /// ```
179    /// use janetrs::{Janet, client::JanetClient, env::DefOptions};
180    /// # fn main() -> Result<(), Box<dyn std::error::Error>> {
181    /// let mut client = JanetClient::init()?;
182    /// assert!(client.env().is_none());
183    ///
184    /// client.add_def(DefOptions::new("const", 10));
185    /// assert!(client.env().is_some());
186    ///
187    /// let c = client.run("const")?;
188    /// assert!(!c.is_nil());
189    /// # Ok(())
190    /// # }
191    /// ```
192    #[inline]
193    pub fn add_def(&mut self, def_opt: DefOptions) {
194        if self.env().is_none() {
195            self.env_table = Some(JanetEnvironment::default());
196        }
197
198        if let Some(ref mut env) = self.env_table {
199            env.add_def(def_opt)
200        }
201    }

267    /// Run given Janet `code` bytes and if no errors occurs, returns the output of the
268    /// given `code`.
269    ///
270    /// **This function may trigger a GC collection**
271    ///
272    /// # Examples
273    /// ```
274    /// use janetrs::{Janet, client::JanetClient};
275    /// # fn main() -> Result<(), Box<dyn std::error::Error>> {
276    /// let client = JanetClient::init_with_default_env()?;
277    ///
278    /// let out = client.run_bytes(b"(def x 10) (+ x x)")?;
279    ///
280    /// assert_eq!(Janet::number(20.0), out);
281    ///
282    /// # Ok(())
283    /// # }
284    /// ```
285    ///
286    ///
287    /// ## TODO:
288    /// Right now the sourcePath value are hardcoded to `c"main"`.
289    /// Change that the Client struct holds sourcePath.
290    #[inline]
291    pub fn run_bytes(&self, code: impl AsRef<[u8]>) -> Result<Janet, Error> {
292        let code = code.as_ref();
293        let env = match self.env_table.as_ref() {
294            Some(e) => e.table(),
295            None => return Err(Error::EnvNotInit),
296        };
297
298        let mut out = Janet::nil();
299
300        let res = unsafe {
301            evil_janet::janet_dobytes(
302                env.raw,
303                code.as_ptr(),
304                code.len() as i32,
305                c"main".as_ptr(),
306                &mut out.inner,
307            )
308        };
309
310        match res {
311            0x01 => Err(Error::RunError),
312            0x02 => Err(Error::CompileError),
313            0x04 => Err(Error::ParseError),
314            _ => Ok(out),
315        }
316    }

318    /// Run given Janet `code` string and if no errors occurs, returns the output of the
319    /// given `code`.
320    ///
321    /// **This function may trigger a GC collection**
322    ///
323    /// # Examples
324    /// ```
325    /// use janetrs::{Janet, client::JanetClient};
326    /// # fn main() -> Result<(), Box<dyn std::error::Error>> {
327    /// let client = JanetClient::init_with_default_env()?;
328    ///
329    /// let out = client.run("(def x 10) (+ x x)")?;
330    ///
331    /// assert_eq!(Janet::number(20.0), out);
332    ///
333    /// # Ok(())
334    /// # }
335    /// ```
336    ///
337    /// ## TODO:
338    /// Right now the sourcePath value are hardcoded to `b"main"`,
339    /// respectively.
340    /// Change that the Client struct hold sourcePath.
341    #[inline]
342    pub fn run(&self, code: impl AsRef<str>) -> Result<Janet, Error> {
343        let code = code.as_ref();
344        self.run_bytes(code.as_bytes())
345    }

347    #[inline]
348    pub fn unmarshal(&self, image: impl AsRef<[u8]>) -> Janet {
349        let image = image.as_ref();
350        let marsh_out = unsafe {
351            use evil_janet::*;
352            let env = janet_core_env(core::ptr::null_mut());
353            let lookup = janet_env_lookup(env);
354            janet_unmarshal(
355                image.as_ptr(),
356                image.len(),
357                0,
358                lookup,
359                core::ptr::null_mut(),
360            )
361        };
362        Janet::from(marsh_out)
363    }

379impl Drop for JanetClient {
380    #[inline]
381    fn drop(&mut self) {
382        // Reset the INIT to false
383        #[cfg(feature = "std")]
384        INIT.with(|i| i.swap(false, Ordering::SeqCst));
385
386        #[cfg(not(feature = "std"))]
387        INIT.swap(false, Ordering::SeqCst);
388
389        unsafe { evil_janet::janet_deinit() }
390    }
391}
